-- Migration 20: past government exams. Each sitting (e.g. 2016 April) is its own section, in order;
-- its questions are normal questions (same bank, same explanations) that also belong to the sitting.
-- Plus the official "emergency & trauma" topic for nursing. Safe to run more than once.
create table if not exists public.exam_sessions (
  id         bigserial primary key,
  track      text not null default 'nursing' references public.tracks(id) on delete cascade,
  year       int not null check (year between 1990 and 2100),
  month      int check (month between 1 and 12),        -- 4 = April, 9 = September ...
  label      text,                                        -- shown name, e.g. "מועד אפריל" / "שחזור"
  note       text,                                        -- where the questions came from (shown to students)
  published  boolean not null default false,
  created_at timestamptz not null default now()
);
create index if not exists exam_sessions_track_idx on public.exam_sessions(track, year, month);
alter table public.exam_sessions enable row level security;
drop policy if exists "read published sessions" on public.exam_sessions;
create policy "read published sessions" on public.exam_sessions for select using (published or public.is_admin());
drop policy if exists "admin sessions" on public.exam_sessions;
create policy "admin sessions" on public.exam_sessions for all using (public.is_admin()) with check (public.is_admin());

alter table public.questions add column if not exists session_id bigint references public.exam_sessions(id) on delete set null;
alter table public.questions add column if not exists session_order int;
create index if not exists questions_session_idx on public.questions(session_id, session_order) where session_id is not null;

-- published sittings of a track that have published questions, newest first
create or replace function public.session_list(p_track text)
returns table (id bigint, year int, month int, label text, note text, n bigint)
language sql stable security definer set search_path = public as $$
  select s.id, s.year, s.month, s.label, s.note, count(q.id)
  from public.exam_sessions s join public.questions q on q.session_id = s.id and q.published
  where s.track = p_track and s.published
  group by s.id order by s.year desc, s.month desc nulls last, s.id desc;
$$;
-- the question ids of one sitting, in the exam's order
create or replace function public.session_ids(p_session bigint)
returns bigint[] language sql stable security definer set search_path = public as $$
  select coalesce(array_agg(q.id order by q.session_order nulls last, q.id), '{}')
  from public.questions q join public.exam_sessions s on s.id = q.session_id
  where q.session_id = p_session and q.published and (s.published or public.is_admin());
$$;
revoke execute on function public.session_list(text) from anon;
revoke execute on function public.session_ids(bigint) from anon;
grant execute on function public.session_list(text) to authenticated;
grant execute on function public.session_ids(bigint) to authenticated;
